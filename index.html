// --- 완전 개편된 SGF 파일 불러오기 기능 --- function loadSGFFile(event) { const file = event.target.files[0]; if (!file) return; const reader = new FileReader(); reader.onload = function(e) { const content = e.target.result; parseSGFAndApply(content); }; reader.readAsText(file); event.target.value = ''; } function parseSGFAndApply(sgfText) { // 1. 바둑판 크기 검출 const szMatch = sgfText.match(/SZ\[(\d+)\]/i); let size = 19; if (szMatch) size = parseInt(szMatch[1], 10); if (size !== 13 && size !== 19) { alert(`SGF의 바둑판 크기(${size}줄)는 지원되지 않는 크기입니다. (19줄 또는 13줄만 가능)`); return; } BOARD_SIZE = size; document.getElementById('boardSizeSelect').value = String(size); updateBoardGeometry(); resetBoard(); // 2. B[...] 또는 W[...] 형태의 착수 패턴을 전체 텍스트에서 순서대로 정규식 추출 // (주석이나 메타데이터, 개행 문자를 무시하고 착수 정보만 직접 수집) const moveRegex = /(B|W)\[([a-zA-Z]{2})?\]/gi; let match; let moveCount = 0; while ((match = moveRegex.exec(sgfText)) !== null) { const colorStr = match[1].toUpperCase(); const coordStr = match[2]; // 패스(Pass)이거나 좌표가 없는 노드는 건너뜀 if (!coordStr || coordStr.length < 2) continue; const color = (colorStr === 'B') ? 1 : 2; const x = sgfCoords.indexOf(coordStr[0].toLowerCase()); const y = sgfCoords.indexOf(coordStr[1].toLowerCase()); // 범위 검증 후 착수 반영 if (x >= 0 && x < BOARD_SIZE && y >= 0 && y < BOARD_SIZE) { const { boardState, deadBlackCount, deadWhiteCount } = getCurrentState(); const opponent = color === 1 ? 2 : 1; const nextBoard = boardState.map(row => [...row]); nextBoard[x][y] = color; let capturedStones = []; getNeighbors(x, y).forEach(n => { if (nextBoard[n.x][n.y] === opponent) { const { group, liberties } = getConnectedGroup(nextBoard, n.x, n.y); if (liberties === 0) capturedStones = capturedStones.concat(group); } }); capturedStones.forEach(pt => { nextBoard[pt.x][pt.y] = 0; }); const addCap = capturedStones.length; const newDeadBlack = deadBlackCount + (color === 2 ? addCap : 0); const newDeadWhite = deadWhiteCount + (color === 1 ? addCap : 0); fullHistory.push({ x, y, color: color, moveNumber: fullHistory.length + 1, boardSnapshot: nextBoard.map(row => [...row]), deadBlackCount: newDeadBlack, deadWhiteCount: newDeadWhite }); moveCount++; } } currentMoveIndex = fullHistory.length; updateStatus(); drawBoard(); if (moveCount === 0) { alert('SGF 파일에서 유효한 기보 착수 정보를 찾지 못했습니다.'); } else { alert(`SGF 기보를 성공적으로 불러왔습니다! (${BOARD_SIZE}줄 / 총 ${fullHistory.length}수)`); } }
